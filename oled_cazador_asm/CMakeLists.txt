cmake_minimum_required(VERSION 3.13)
if(NOT PICO_BOARD)
    set(PICO_BOARD waveshare_rp2040_zero CACHE STRING "RP2040 board")
endif()
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/pico_sdk_init.cmake")
    message(FATAL_ERROR "Indica PICO_SDK_PATH o usa el CMake generado por la extension Pico")
endif()
include("${PICO_SDK_PATH}/pico_sdk_init.cmake")
project(oled_cazador_asm C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(oled_cazador_asm oled_cazador_asm.c programa.S)
target_link_libraries(oled_cazador_asm pico_stdlib hardware_adc hardware_i2c)
pico_enable_stdio_usb(oled_cazador_asm 1)
pico_enable_stdio_uart(oled_cazador_asm 0)
pico_add_extra_outputs(oled_cazador_asm)
